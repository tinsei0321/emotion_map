    ```javascript
      if (!ctx.extracted || !ctx.extracted.geo || !ctx.extracted.geo.length) return null;  // 三条件·空 geo 不衔接
      const q = ctx.question || '';
      if (/标记|标到地图|在地图上|点位|把.*标/.test(q)) return 'markup';
      if (/分析|密度|分布|热力|聚集|集中/.test(q)) return 'analyze';
      if (/对比|比较|VS|vs/.test(q)) return 'compare';
      if (/归因|为什么|最差|最好|排序/.test(q)) return 'attribute';
      if (/裁出|周边|附近.*米/.test(q)) return 'extract';
      return null;
    }
    ```

#### 悬点 ③：首问防误衔接（**claude plan 悬点·glm 补代码级守卫**）

- **场景**：首问"分析葛洲坝片区"（非追问·无 priorTurn）。
